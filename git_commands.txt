git init-создаёт новый пустой репозиторий в текущей папке
git clone <url>копируетдалённый репозиторий к себе на компьютер
git status – показывает, какие файлы изменены или готовы к коммиту
git add <файл>           добавляет файл в «зону подготовки» 
git commit -m "текст"– сохраняет добавленные изменения с пояснением, что сделано
git push  – отправляет твои коммиты на удалённый сервер 
git pull  – забирает свежие изменения с сервера и объединяет с твоими
git log   – показывает историю всех коммитов 
git diff -показывает, что именно изменилось в файлах с последнего коммита
git branch список свех веток репозитории
git checkout <ветка> переключается на другую ветку
git merge <ветка> вливает изменения из указанной ветки в текущую
git remote -v - показывает адрес удаленых репозиториев
